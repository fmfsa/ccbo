\section{Background}
\label{sec:background}

% CONTENT (≤0.5 page total; heavy citation, minimal prose).

\paragraph{SCMs.}
% CONTENT: one sentence defining $\mathcal{M}=\langle \mathbf{U},\mathbf{V},\mathcal{F},P(\mathbf{U})\rangle$ and $\do(X_S=\mathbf{x}_S)$; cite Pearl 2009.

\paragraph{Causal Bayesian Optimization.}
% CONTENT: one sentence — CBO optimises $\E[Y \mid \do(X_S=\mathbf{x}_S)]$ over non-dominated intervention sets (MIS/POMIS) with causal GP priors; cite Aglietti 2020 and Garnett 2023.
\begin{equation}
  \label{eq:cbo_obj}
  (S^*, \mathbf{x}^*) = \arg\min_{\substack{S \in \mathfrak{S} \\ \mathbf{x}_S \in \mathcal{X}_S}}
  \E\!\left[Y \mid \do(X_S = \mathbf{x}_S)\right].
\end{equation}

\paragraph{Coarsening and C-DAGs.}
% CONTENT: one sentence — partition $\pi$ is valid iff $\Gq$ is acyclic; cite Madaleno 2026, Anand 2023 (soundness of do-calculus on C-DAGs), Wahl 2024 ($\sigma$-Markov transfers, faithfulness does not).

\begin{definition}[Coarsening~\citep{madaleno2026coarsening}]
\label{def:coarsening}
Given a DAG $\Gg$ on $\mathbf{V}$, a \textbf{coarsening} is a partition $\pi=\{C_1,\ldots,C_{d'}\}$ such that the induced quotient $\Gq$, with nodes $\{C_1,\ldots,C_{d'}\}$ and $C_k\to C_\ell$ iff $\exists V_i\in C_k, V_j\in C_\ell$ with $V_i\to V_j$ in $\Gg$, is a DAG.
\end{definition}

\subsection{Two-tier causal GP prior}
\label{sec:causal_prior}

% CONTENT (one short paragraph): identification is performed on $\Gq$ (not $\Gg$) using the complete ID algorithm \citep{shpitser2006identification} via \texttt{ananke-causal} on the projected ADMG. When ID returns \textsc{NotIdentified}, we fall back to the observational mean $\bar{Y}_{\mathrm{obs}}$ (uninformative prior); the BO loop then learns the interventional effect purely from experimental data. Cite Pearl 1995/2009, Anand 2023, Tian-Pearl 2002.

\begin{equation}
  \label{eq:two_tier_prior}
  \mu_0^S(\mathbf{x}_S) = \begin{cases}
    \hat{\E}[Y \mid \do(X_S = \mathbf{x}_S)] & \text{if identifiable from } \Gq, \\
    \bar{Y}_{\mathrm{obs}} & \text{otherwise.}
  \end{cases}
\end{equation}

\subsection{Assumptions}
\label{sec:assumptions}

% CONTENT: this is the honest methodological-caveats box; three bullets.
\begin{assumption}[Valid partition]
\label{ass:valid}
% CONTENT: the practitioner supplies a partition $\pi$ whose quotient $\Gq$ is acyclic.
\end{assumption}
\begin{assumption}[Cluster-level identification as subset prior]
\label{ass:cluster_prior}
% CONTENT: identification is derived for $\do(C_k)$ (cluster-level; Anand 2023) and re-used as a GP prior mean for subset interventions $\do(C_k\cap\mathbf{X})$. This is a heuristic — not a formal ID guarantee for sub-cluster do's.
\end{assumption}
\begin{assumption}[GID-PO completeness boundary]
\label{ass:id_scheme}
% CONTENT: we use the Shpitser-Pearl ID algorithm~\citep{shpitser2006identification} on the projected ADMG $\Gq$, implemented via \texttt{ananke-causal}. ID is sound and complete for non-parametric identification from $\Gq$; non-identified queries return a \textsc{NotIdentified} exception and fall back to uninformative prior. This is a principled — not heuristic — fallback.
\end{assumption}

\subsection{Algorithm}

\begin{algorithm}[t]
\caption{\CCBO: Coarsened Causal Bayesian Optimization}
\label{alg:ccbo}
\begin{algorithmic}[1]
\REQUIRE $\Gq$ (ADMG with bidirected edges for hidden confounders); $\pi$; target $Y$; manipulative set $\mathbf{X}$; budget $T$; observational data.
\STATE Compute $\mathbf{X}^\pi$ and exploration set $\mathfrak{S}^{\pi}$ via Eq.~\eqref{eq:ccbo_exploration}.
\STATE For each $S \in \mathfrak{S}^{\pi}$, call \textsc{ID} (Shpitser \& Pearl 2006) on $\Gq$ via \texttt{ananke-causal}; compute $\mu_0^S$ via Eq.~\eqref{eq:two_tier_prior}; fall back to $\bar{Y}_{\mathrm{obs}}$ if \textsc{NotIdentified}.
\STATE Collect initial interventional data $\mathcal{D}_0$.
\FOR{$t = 1, \ldots, T$}
  \STATE Update GP posteriors $(\mu_t^S, k_t^S)$ for all $S \in \mathfrak{S}^{\pi}$.
  \STATE $(S_t, \mathbf{x}_t) \leftarrow \arg\max_{S,\mathbf{x}} \mathrm{EI}_{S,t}(\mathbf{x})$.
  \STATE Observe $y_t$; update $\mathcal{D}_t$.
\ENDFOR
\RETURN $\mathbf{x}_{t^*}$ where $t^* = \arg\min_t y_t$.
\end{algorithmic}
\end{algorithm}

\section{Theoretical properties}
\label{sec:theory}

% CONTENT (≤0.5 page): three propositions, foreground Prop. 2.

\begin{proposition}[Identity recovery]
\label{prop:identity}
When $\pi=\pi_\top$, \CCBO{} reduces to \CBO{} with power-set exploration.
\end{proposition}
% CONTENT: one-line proof — identity partition gives singleton clusters, so $\mathfrak{S}^{\pi_\top}$ is the full power set and $\Gq=\Gg$.

\begin{proposition}[Identification invariance under intra-cluster edges]
\label{prop:invariance}
Let $\pi$ be a valid coarsening of $\Gg$ and $e=(V_i\to V_j)$ with $V_i,V_j\in C_k\in\pi$. Then (i) $\Gq=(\Gg\setminus e)^{\pi}$, and (ii) for every $S\in\mathfrak{S}^\pi$, the identification formula for $\E[Y\mid\do(X_S=\mathbf{x}_S)]$ derived from $\Gq$ is the same under $\Gg$ and $\Gg\setminus e$.
\end{proposition}
\begin{proof}
% CONTENT: 3 lines — inter-cluster edges unaffected by removing an intra-cluster $e$; $\Gq$ identical; every downstream object (ES, adjustment set, ID formula) depends only on $\Gq$.
\end{proof}

\begin{remark}[Structural robustness]
\label{rem:structural_robustness}
% CONTENT: one short paragraph foregrounding Prop. 2 — intra-cluster misspecification is invisible; this is the paper's core insight.
\end{remark}

\begin{proposition}[Refinement monotonicity]
\label{prop:refinement}
Let $\pi'$ refine $\pi$. Then (i) $\pi'$ is valid; (ii) $\mathfrak{S}^{\pi'}\supseteq\mathfrak{S}^\pi$ in the sense that every entry in $\mathfrak{S}^\pi$ maps to at least one entry in $\mathfrak{S}^{\pi'}$; (iii) every effect identifiable from $\Gq$ remains identifiable from $\mathcal{G}^{\pi'}$.
\end{proposition}
\begin{proof}
% CONTENT: 4 lines — cycles can't form under refinement; splitting a cluster can only add ES entries; refinement can only remove bidirected edges, never add them.
\end{proof}

\section{Recursive CCBO (teaser)}
\label{sec:rccbo}

% CONTENT (≤1 page total): one-paragraph motivation — BO $\to$ CCBO $\to$ CBO as evidence accumulates; refinement via RePaRe on interventional data; by Prop. \ref{prop:refinement} every refinement is monotonically safe. Pointer to \S\ref{sec:rccbo_exp}.

\begin{algorithm}[t]
\caption{\RCCBO: Recursive Coarsened Causal Bayesian Optimization}
\label{alg:rccbo}
\begin{algorithmic}[1]
\REQUIRE Manipulative $\mathbf{X}$, $\mathcal{D}_\text{obs}$, budget $T$, phase $k$, RePaRe levels $\alpha,\beta$.
\STATE $\pi\leftarrow\text{None}$; $\mathfrak{S}\leftarrow\{\{x\}:x\in\mathbf{X}\}$; $\mathcal{D}_\text{int}\leftarrow\emptyset$.
\WHILE{budget remaining}
  \IF{$\pi\neq\text{None}$} \STATE Build $\Gq$; compute $\mathfrak{S}^\pi$ and two-tier priors \ELSE \STATE Uninformative priors \ENDIF
  \STATE Run $\min(k,\text{rem.})$ trials of \CBO{} on $\mathfrak{S}$; record full-variable samples into $\mathcal{D}_\text{int}$.
  \STATE $\pi'\leftarrow\textsc{RePaRe}(\mathcal{D}_\text{obs},\mathcal{D}_\text{int};\alpha,\beta)$.
  \IF{$\pi'\neq\pi$} \STATE Patch GP state; $\pi\leftarrow\pi'$. \ENDIF
\ENDWHILE
\RETURN $\mathbf{x}_{t^*}$, final $\pi$.
\end{algorithmic}
\end{algorithm}
