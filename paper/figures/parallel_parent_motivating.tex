% Motivating example (Fig. 1): the ParallelParent SCM as three subfigures.
% (a) true fine graph; (b) misspecified fine graph with X1->Y deleted;
% (c) the common quotient under C1 = {X1, X2}.
% Uses the shared DAG styles from the main preamble (mannode/tgtnode/clbox/
% diredge/bicross) and the subcaption package loaded there. Hand-written:
% this is a one-off 3-node figure; the registered-graph gallery in the
% appendix is emitted by scripts/emit_dataset_tikz.py.
\begin{subfigure}[b]{0.32\linewidth}
  \centering
  \begin{tikzpicture}[x=1cm,y=1cm]
    \node[mannode] (X1a) at (0,1.15) {$X_1$};
    \node[mannode] (X2a) at (1.4,1.15) {$X_2$};
    \node[tgtnode] (Ya)  at (0.7,-0.35) {$Y$};
    \draw[diredge] (X1a) -- (Ya);
    \draw[diredge] (X2a) -- (Ya);
    \draw[bicross] (X1a) to[bend left=28] (X2a);
  \end{tikzpicture}
  \caption{True graph}
  \label{fig:motivating-true}
\end{subfigure}\hfill
\begin{subfigure}[b]{0.32\linewidth}
  \centering
  \begin{tikzpicture}[x=1cm,y=1cm]
    \node[mannode] (X1b) at (0,1.15) {$X_1$};
    \node[mannode] (X2b) at (1.4,1.15) {$X_2$};
    \node[tgtnode] (Yb)  at (0.7,-0.35) {$Y$};
    % the deleted edge, ghosted with a strike mark
    \draw[diredge,black!25] (X1b) -- (Yb)
      node[midway,sloped] {\textcolor{red!70!black}{\footnotesize$\boldsymbol{\times}$}};
    \draw[diredge] (X2b) -- (Yb);
    \draw[bicross] (X1b) to[bend left=28] (X2b);
  \end{tikzpicture}
  \caption{Assumed graph}
  \label{fig:motivating-assumed}
\end{subfigure}\hfill
\begin{subfigure}[b]{0.32\linewidth}

      \node[clbox,inner sep=2.6mm,fit=(X1c)(X2c),

  \end{tikzpicture}
  \caption{Common quotient}
