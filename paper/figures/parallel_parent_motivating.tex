% Motivating example (Fig. 1): the ParallelParent SCM.
% (a) true fine graph; (b) misspecified fine graph with X1->Y deleted;
% (c) the common quotient under C1 = {X1, X2}.
% Uses the shared DAG styles from the main preamble (mannode/tgtnode/clbox/
% diredge/bicross). Hand-written: this is a one-off 3-node figure; the
% registered-graph gallery in App. D is emitted by scripts/emit_dataset_tikz.py.
\begin{tikzpicture}[x=1cm,y=1cm,
    panel/.style={font=\scriptsize\itshape,anchor=north}]
  % ---------------- (a) true fine graph -------------------------------------
  \begin{scope}[shift={(0,0)}]
    \node[mannode] (X1a) at (0,1.15) {$X_1$};
    \node[mannode] (X2a) at (1.5,1.15) {$X_2$};
    \node[tgtnode] (Ya)  at (0.75,-0.35) {$Y$};
    \draw[diredge] (X1a) -- (Ya);
    \draw[diredge] (X2a) -- (Ya);
    \draw[bicross] (X1a) to[bend left=28] (X2a);
    \node[panel] at (0.75,-0.95) {(a) true $\Gg^{*}$};
  \end{scope}
  % ---------------- (b) wrong fine graph ------------------------------------
  \begin{scope}[shift={(3.1,0)}]
    \node[mannode] (X1b) at (0,1.15) {$X_1$};
    \node[mannode] (X2b) at (1.5,1.15) {$X_2$};
    \node[tgtnode] (Yb)  at (0.75,-0.35) {$Y$};
    % the deleted edge, ghosted with a strike mark
    \draw[diredge,black!25] (X1b) -- (Yb)
      node[midway,sloped] {\textcolor{red!70!black}{\footnotesize$\boldsymbol{\times}$}};
    \draw[diredge] (X2b) -- (Yb);
    \draw[bicross] (X1b) to[bend left=28] (X2b);
    \node[panel] at (0.75,-0.95) {(b) assumed $\tilde{\Gg}$};
  \end{scope}
  % ---------------- (c) the common quotient ---------------------------------
  \begin{scope}[shift={(6.2,0)}]
    \node[mannode] (X1c) at (0,1.15) {$X_1$};
    \node[mannode] (X2c) at (1.5,1.15) {$X_2$};
    \node[tgtnode] (Yc)  at (0.75,-0.35) {$Y$};
    \begin{scope}[on background layer]
      \node[clbox,fit=(X1c)(X2c),label={[font=\scriptsize]above:$C_1$}] (C1c) {};
    \end{scope}
    \draw[diredge] (C1c.south) -- (Yc);
    \node[panel] at (0.75,-0.95) {(c) quotient $\Gq$ of both};
  \end{scope}
  % ---------------- quotient braces -----------------------------------------
  \draw[->,>=Stealth,black!45,shorten >=2pt,shorten <=2pt]
    (2.05,0.4) -- (2.65,0.4);
  \node[font=\scriptsize,black!60] at (2.35,0.72) {del};
  \draw[->,>=Stealth,black!45,shorten >=2pt,shorten <=2pt]
    (5.15,0.4) -- (5.75,0.4);
  \node[font=\scriptsize,black!60] at (5.45,0.72) {$/\Pi$};
\end{tikzpicture}
